// CÃLCULO DE DANO COM FATOR DE CRÃTICO E DEFESA
// -------------------------------------------------------------
function calc_damage(_atk, _def, _crit_chance, _crit_rate, _is_defending) {
    var _base_dmg = max(1, _atk - (_def / 2));
    
    // Teste de Acerto CrÃ­tico
    var _is_crit = (random(100) < _crit_chance);
    if (_is_crit) {
        _base_dmg *= (1 + (_crit_rate / 100));
    }
    
    // ReduÃ§Ã£o de 50% se o alvo estiver defendendo
    if (_is_defending) _base_dmg = max(1, _base_dmg * 0.5);

    return { damage: round(_base_dmg), is_crit: _is_crit };
}

// -------------------------------------------------------------
// AÃ‡Ã•ES DO JOGADOR
// -------------------------------------------------------------
function player_attack() {
    if (state != BATTLE_STATE.PLAYER_TURN) exit;
    player_defending = false;

    var _p = Obj_Player.stats;
    var _res = calc_damage(_p.attack, enemy_defense, _p.crit_chance, _p.crit_rate, enemy_defending);

    enemy_hp = max(0, enemy_hp - _res.damage);
    
    var _msg = _p.name + " atacou causa " + string(_res.damage) + " de dano!";
    if (_res.is_crit) _msg += " (CRÃTICO!)";
    add_log(_msg);

    check_battle_end();
}

function player_defend() {
    if (state != BATTLE_STATE.PLAYER_TURN) exit;
    player_defending = true;
    
    var _p = Obj_Player.stats;
    add_log(_p.name + " assumiu posiÃ§Ã£o de defesa!");
    
    state = BATTLE_STATE.ENEMY_TURN;
}

function player_use_skill(_skill) {
    if (state != BATTLE_STATE.PLAYER_TURN) exit;
    var _p = Obj_Player.stats;

    if (_p.mp < _skill.mp) {
        add_log("MP insuficiente para " + _skill.name + "!");
        exit;
    }

    _p.mp -= _skill.mp;
    player_defending = false;

    var _res = calc_damage(_p.magic + (_skill[$ "power"] ?? 5), enemy_defense, _p.crit_chance, _p.crit_rate, enemy_defending);
    enemy_hp = max(0, enemy_hp - _res.damage);

    add_log(_p.name + " usou " + _skill.name + " causando " + string(_res.damage) + " de dano!");
    check_battle_end();
}

function player_use_item(_slot_idx) {
    if (state != BATTLE_STATE.PLAYER_TURN) exit;
    if (!instance_exists(Obj_HUD)) exit;

    var _item = Obj_HUD.inv_grid[_slot_idx];
    if (_item == undefined || _item.type != ITEM_TYPE.CONSUMABLE) {
        add_log("Este item nÃ£o pode ser usado em batalha!");
        exit;
    }

    var _p = Obj_Player.stats;
    player_defending = false;

    if (_item.hp_rec > 0) {
        _p.life = min(_p.life_max, _p.life + _item.hp_rec);
        add_log(_p.name + " usou " + _item.name + " e recuperou " + string(_item.hp_rec) + " HP!");
    }
    if (_item.mp_rec > 0) {
        _p.mp = min(_p.mp_max, _p.mp + _item.mp_rec);
        add_log(_p.name + " usou " + _item.name + " e recuperou " + string(_item.mp_rec) + " MP!");
    }

    // Consome o item do inventÃ¡rio
    Obj_HUD.inv_grid[_slot_idx] = undefined;
    
    state = BATTLE_STATE.ENEMY_TURN;
}

function check_battle_end() {
    if (enemy_hp <= 0) {
        state = BATTLE_STATE.VICTORY;
		
		// TROCA IMEDIATAMENTE O SPRITE E ESTADO DO INIMIGO NO MAPA 3D
        if (variable_global_exists("active_enemy") && instance_exists(global.active_enemy)) {
            with (global.active_enemy) {
                is_dead = true;
                
                // Aplica o sprite de cadÃ¡ver assim que o HP zera
                if (variable_instance_exists(id, "sprite_dead") && sprite_exists(Spr_Enemy_Dead)) {
                    sprite_index = sprite_dead;
                }
            }
        }
		
	    add_log("--- VITÃ“RIA! ---");
	    add_log("VocÃª derrotou " + enemy_name + "!");

	    // Sorteia Drops e XP
	    xp_gained = enemy_xp;
	    add_log("Ganhou " + string(xp_gained) + " de XP!");

	    // Chama a funÃ§Ã£o global que recalcula a fÃ³rmula de Level Up
	    player_gain_xp(xp_gained);

        if (script_exists(asset_get_index("generate_loot"))) {
            victory_loot = generate_loot(enemy_data_ref);
            for (var i = 0; i < array_length(victory_loot); i++) {
                var _drop = victory_loot[i];
                add_log("Dropou: " + _drop.name);
                
                // Adiciona ao primeiro slot livre do inventÃ¡rio
                for (var s = 0; s < array_length(Obj_HUD.inv_grid); s++) {
                    if (Obj_HUD.inv_grid[s] == undefined) {
                        Obj_HUD.inv_grid[s] = _drop;
                        break;
                    }
                }
            }
        }
    } else {
        state = BATTLE_STATE.ENEMY_TURN;
    }
}

/// @desc Cria o struct do player a partir da classe escolhida. Sorteia o Mantra aqui.
/// @param {Real} _class_id  valor do enum CLASS
/// @param {String} _name    nome digitado
function create_player(_class_id, _name) {
    if (!variable_global_exists("classes")) init_classes();

    var _c = global.classes[_class_id];
    var _is_mantra = (random(100) < _c.mantra_chance);
    var _mult = _is_mantra ? _c.mantra_cost_mult : 1;
	
	var _initial_level = 1;
    var _base_xp = _c[$ "base_xp"] ?? 100;
	
	// FÃ“RMULA: XP_next = Base * (Level ^ 1.5)
    var _xp_next = round(_base_xp * power(_initial_level, 1.5));

    var _skills = [];
    for (var i = 0; i < array_length(_c.skills); i++) {
        var _s = _c.skills[i];
        array_push(_skills, { name: _s.name, mp: ceil(_s.mp * _mult), desc: _s.desc });
    }
    if (_is_mantra) {
        for (var j = 0; j < array_length(_c.mantra_skills); j++) {
            var _m = _c.mantra_skills[j];
            array_push(_skills, { name: _m.name, mp: _m.mp, desc: _m.desc });
        }
    }

    return {
        name: _name,
        class_id: _class_id,
        class_name: _c.name,
        portrait: _c.player_portrait, 
        is_mantra: _is_mantra,
        level: 1,
		
		// SISTEMA DE EXPERIÃŠNCIA
        base_xp: _base_xp,
        xp: 0,
        xp_next: _xp_next,

        life: _c.life,  life_max: _c.life,
        mp: _c.mp,      mp_max: _c.mp,
        xp: _c.xp,
        attack: _c.attack,
        defense: _c.defense,
        magic: _c.magic,
        crit_chance: _c.crit_chance,
        crit_rate: _c.crit_rate,
		
		// GANHOS DE ATRIBUTOS POR LEVEL UP
        hp_gain:  _c[$ "hp_gain"]  ?? 5,
        mp_gain:  _c[$ "mp_gain"]  ?? 5,
        atk_gain: _c[$ "atk_gain"] ?? 2,
        def_gain: _c[$ "def_gain"] ?? 1,
        mag_gain: _c[$ "mag_gain"] ?? 1,

        skills: _skills
    };
}

/// @desc Adiciona XP ao jogador e processa a elevaÃ§Ã£o de nÃ­vel
/// @param {Real} _amount Quantidade de experiÃªncia recebida
function player_gain_xp(_amount) {
    if (!instance_exists(Obj_Player)) exit;
    var _p = Obj_Player.stats;

    _p.xp += _amount;

    // Loop para processar acÃºmulos que sobem mais de 1 nÃ­vel de uma vez
    while (_p.xp >= _p.xp_next) {
        // Deduz a meta cumprida mantendo o excesso de XP
        _p.xp -= _p.xp_next;
        _p.level += 1;

        // FÃ“RMULA: XP_next = Base * (Level ^ 1.5)
        _p.xp_next = round(_p.base_xp * power(_p.level, 1.5));

        // Incrementa os atributos da classe
        _p.life_max += _p.hp_gain;
        _p.life      = _p.life_max; // Restaura HP totalmente ao subir de nÃ­vel
        
        _p.mp_max   += _p.mp_gain;
        _p.mp        = _p.mp_max;   // Restaura MP totalmente ao subir de nÃ­vel

        _p.attack   += _p.atk_gain;
        _p.defense  += _p.def_gain;
        _p.magic    += _p.mag_gain;

        // NotificaÃ§Ã£o de Level Up no log da batalha (se ativo)
        if (instance_exists(Obj_Battle)) {
            Obj_Battle.add_log("--- LEVEL UP! ---");
            Obj_Battle.add_log(_p.name + " subiu para o NÃ­vel " + string(_p.level) + "!");
        }
    }
}

    // 4. Direita (X2)
    _add_quad(_vb, x2, y1, z2,  x2, y2, z2,  x2, y2, z1,  x2, y1, z1, _h, _v);
    // 5. Teto (Z2)
    _add_quad(_vb, x1, y1, z2,  x2, y1, z2,  x2, y2, z2,  x1, y2, z2, _h, _v);
    // 6. ChÃ£o (Z1)
    _add_quad(_vb, x1, y2, z1,  x2, y2, z1,  x2, y1, z1,  x1, y1, z1, _h, _v);

    vertex_end(_vb);
    vertex_submit(_vb, pr_trianglelist, tex);
    vertex_delete_buffer(_vb);
}

/// @desc Desenha uma caixa 3D em wireframe (Bounding Box de depuraÃ§Ã£o)
/// @param {Real} _x PosiÃ§Ã£o X no mundo
/// @param {Real} _y PosiÃ§Ã£o Y no mundo
/// @param {Real} _z PosiÃ§Ã£o Z (ElevaÃ§Ã£o do centro da caixa)
/// @param {Real} _w Largura (Tamanho X)
/// @param {Real} _h Comprimento (Tamanho Y)
/// @param {Real} _d Altura (Tamanho Z)
/// @param {Constant.Color} _color Cor das linhas
function d3d_draw_wireframe_box(_x, _y, _z, _w, _h, _d, _color) {
    // Define a matriz de transformaÃ§Ã£o centrada nas coordenadas do objeto
    var _matrix = matrix_build(_x, _y, _z, 0, 0, 0, 1, 1, 1);
    matrix_set(matrix_world, _matrix);

    // Garante que o Z-Buffer esteja ativo
    gpu_set_zwriteenable(true);
    gpu_set_ztestenable(true);

    // Metades das dimensÃµes para centralizar o cubo na origem (0,0,0)
    var _hw = _w / 2;
    var _hh = _h / 2;
    var _hd = _d / 2;

    // Inicializa o formato de vÃ©rtice (PosiÃ§Ã£o 3D + Cor)
    static format = undefined;
    if (format == undefined) {
        vertex_format_begin();
        vertex_format_add_position_3d();
        vertex_format_add_color();
        format = vertex_format_end();
    }

    var _vbuf = vertex_create_buffer();
    vertex_begin(_vbuf, format);

    // FunÃ§Ã£o interna para registrar cada aresta (linha entre dois pontos)
    var _add_line = function(_vb, _x1, _y1, _z1, _x2, _y2, _z2, _col) {
        vertex_position_3d(_vb, _x1, _y1, _z1);
        vertex_color(_vb, _col, 1);
        vertex_position_3d(_vb, _x2, _y2, _z2);
        vertex_color(_vb, _col, 1);
    };

    // 1. Base Inferior (Z = -_hd)
    _add_line(_vbuf, -_hw, -_hh, -_hd,  _hw, -_hh, -_hd, _color);
    _add_line(_vbuf,  _hw, -_hh, -_hd,  _hw,  _hh, -_hd, _color);
    _add_line(_vbuf,  _hw,  _hh, -_hd, -_hw,  _hh, -_hd, _color);
    _add_line(_vbuf, -_hw,  _hh, -_hd, -_hw, -_hh, -_hd, _color);

    // 2. Base Superior (Z = +_hd)
    _add_line(_vbuf, -_hw, -_hh,  _hd,  _hw, -_hh,  _hd, _color);
    _add_line(_vbuf,  _hw, -_hh,  _hd,  _hw,  _hh,  _hd, _color);
    _add_line(_vbuf,  _hw,  _hh,  _hd, -_hw,  _hh,  _hd, _color);
    _add_line(_vbuf, -_hw,  _hh,  _hd, -_hw, -_hh,  _hd, _color);

    // 3. Colunas Verticais
    _add_line(_vbuf, -_hw, -_hh, -_hd, -_hw, -_hh,  _hd, _color);
    _add_line(_vbuf,  _hw, -_hh, -_hd,  _hw, -_hh,  _hd, _color);
    _add_line(_vbuf,  _hw,  _hh, -_hd,  _hw,  _hh,  _hd, _color);
    _add_line(_vbuf, -_hw,  _hh, -_hd, -_hw,  _hh,  _hd, _color);

    vertex_end(_vbuf);
    
    // Submete a geometria de linhas para a GPU (Sem textura)
    vertex_submit(_vbuf, pr_linelist, -1);
    vertex_delete_buffer(_vbuf);

    // Reseta a matriz de mundo para a identidade
    matrix_set(matrix_world, matrix_build_identity());
}

/// @desc Desacelera suavemente no final (Excelente para placas de UI)
function ease_out_cubic(t) {
    t = clamp(t, 0, 1);
    return 1 - power(1 - t, 3);
}

/// @desc Acelera no inÃ­cio e desacelera no final (Curva em S clÃ¡ssica)
function smoothstep(t) {
    t = clamp(t, 0, 1);
    return t * t * (3 - 2 * t);
}

/// @desc Processa todas as chances de drop de uma struct de inimigo/baÃº e retorna um array com os itens sorteados
/// @param {Struct} _drop_data A struct do inimigo (ex: global.enemies[ENEMY.DESGRACA])
/// @return {Array} Array de structs de itens dropados
function generate_loot(_drop_data) {
    if (!variable_global_exists("items")) init_items();

    var _dropped_items = [];

    // Mapeia os campos do seu script para os pools de itens
    var _categories = [
        { chance: _drop_data.potion_heal_drop_chance,   pool: "potion_heal" },
        { chance: _drop_data.potion_mana_drop_chance,   pool: "potion_mana" },
        { chance: _drop_data.key_drop_chance,           pool: "key" },
        { chance: _drop_data.file_drop_chance,          pool: "file" },
        { chance: _drop_data.file_special_drop_chance,  pool: "file_special" },
        { chance: _drop_data.equip_drop_chance,         pool: "equip" },
        { chance: _drop_data.equip_special_drop_chance, pool: "equip_special" }
    ];

    // Testa cada uma das 7 porcentagens de drop do inimigo
    for (var i = 0; i < array_length(_categories); i++) {
        var _cat = _categories[i];
        
        // Trata valores especiais como 'infinity' (Fagulha) ou sorteios comuns de 0 a 100
        var _passed = false;
        if (_cat.chance == infinity) {
            _passed = true;
        } else if (_cat.chance > 0) {
            _passed = (random(100) < _cat.chance);
        }

        // Se passou no teste de porcentagem, escolhe um item aleatÃ³rio daquela categoria
        if (_passed) {
            var _item_list = global.item_pools[$ _cat.pool];
            if (array_length(_item_list) > 0) {
                var _chosen_id = _item_list[irandom(array_length(_item_list) - 1)];
                array_push(_dropped_items, global.items[$ _chosen_id]);
            }
        }
    }

    return _dropped_items;
}

function move_player(){
	var up = keyboard_check_pressed(vk_up);
	var down = keyboard_check_pressed(vk_down);
	var left = keyboard_check_pressed(vk_left);
	var right = keyboard_check_pressed(vk_right);

	if (up)
	{
	    // Move Forward
    
	    var vect_x = lengthdir_x(32, target_dir);
	    var vect_y = lengthdir_y(32, target_dir);
		
	    var _target_gx = vect_x div Obj_Builder.cell_width;
		var _target_gy = vect_y div Obj_Builder.cell_height;
    
	    // Move Forward
	    if (Obj_Builder.grid[# _target_gx, _target_gy] != Obj_Builder.CELL_WALL)
	    {
	        target_x += vect_x;
	        target_y += vect_y;
	    }
	}
	if (down)
	{
	    // Move Backwards
    
	    var vect_x = lengthdir_x(32, target_dir);
	    var vect_y = lengthdir_y(32, target_dir);
		
		var _target_gx = vect_x div Obj_Builder.cell_width;
		var _target_gy = vect_y div Obj_Builder.cell_height;
    
	    // Move Forward
	    if (Obj_Builder.grid[# _target_gx, _target_gy] != Obj_Builder.CELL_WALL)
	    {
	        target_x -= vect_x;
	        target_y -= vect_y;
	    }
	}
	if (left)
	{
	    // Turn Left
	    target_dir -= 90;
	}
	if (right)
	{
	    // Turn Right
	    target_dir += 90;
	}
}
